% Late Night Pal look, shared by document.tex and cheat-sheet.tex (Python-owned; loaded before the guard).
% The generated body may only use the environments and macros below; tcolorbox commands themselves are
% rejected by validation.latex_problems, so a body can't pass its own box options.

\usepackage{titlesec}

% Headings

\titleformat{\paragraph}[runin]{\normalsize\bfseries\color{lnpblue}}{}{0pt}{}

    lnpbreak, lnpspace}}
