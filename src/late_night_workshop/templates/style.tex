% Headings: numbered sections are solid bars; subsections are purple sans lines.

\titleformat{\section}{\lnpsans\bfseries\large\color{white}}{}{0pt}{\lnpbar{lnpbar}{\thesection\hspace{1.3em}}}
\titleformat{name=\section,numberless}{\lnpsans\bfseries\large\color{white}}{}{0pt}{\lnpbar{lnpbar}{}}

% Purple bar for practice sets, green bar for answers: \practicesection{Practice · Multiple choice}
